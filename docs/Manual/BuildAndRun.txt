   > brew install kubectl

   > brew install weaveworks/tap/eksctl

   > brew install helm

   > brew install mysql-client@8.4 # couldn't connect to MySQL (8.0.43) RDS instance when using mysql-client 9.5.0

   > tfenv use 1.9.8  # Switch to that version

##########
## STEP 3 - Prepare subnets and SGs
##########
 - Create a VPC
 - Create subnets as specified in the NW design
 - Attach IGW to your VPC
 - Create NGWs in your public subnet
 - Create route tables for
    - public subnet (subnet with 0.0.0.0/0 -> IGW) for ALB to allow internet access
    - private subnet with 0.0.0.0/0 -> NGW for EKS worker nodes
    (EKS cluster node and DB only use default route table)
 - Attach route tables
 - Create SGs for
    - ALB: only allow inbound traffic from intended users
           tag kubernetes.io/role/elb=1 and kubernetes.io/cluster/graphapp-eks-cluster=owned
           to let EKS controller find public subnets automatically
    - worker nodes: as specified in the NW design xlsx
    - control plane ENIs: as specified in the NW design xlsx
    - RDS: allow DB access (3306/tcp for MySQL) to the SG of worker nodes

## STEP 5 - Prepare Container Images
##########
You can skip if ECR private repositories for frontend and backend already exist.
```
# Create Images (If not already created)
bash scripts/build_images.sh

# Create repositories
aws ecr create-repository --repository-name graphapp-eks-frontend
aws ecr create-repository --repository-name graphapp-eks-backend

# Login Docker to ECR
aws ecr get-login-password \
  | docker login --username AWS --password-stdin <AWS_ACCOUNT_ID>.dkr.ecr.<REGION>.amazonaws.com

# Tag images (Local Docker must be running)
docker tag frontend:latest <ACCOUNT_ID>.dkr.ecr.<REGION>.amazonaws.com/my-frontend:latest
docker tag backend:latest <ACCOUNT_ID>.dkr.ecr.<REGION>.amazonaws.com/my-backend:latest

# Push images
docker push <ACCOUNT_ID>.dkr.ecr.<REGION>.amazonaws.com/my-frontend:latest
docker push <ACCOUNT_ID>.dkr.ecr.<REGION>.amazonaws.com/my-backend:latest
```

## STEP 7 Configure Ingress / ALB / Node AutoScaler
##########
 -Your Ingress controller will create an ALB endpoint.
```
# # Install the CRD of ALB (TargetGroupBinding: A Kubernetes Object is created that represents ALB instances)
# kubectl apply -k github.com/aws/eks-charts/stable/aws-load-balancer-controller/crds?ref=master
# This is automatically done by EKS so coommented out

# Add Helm repo
helm repo add eks https://aws.github.io/eks-charts
helm repo update

# Enable OIDC provider (Without OIDC IRSA won't work: disabled by default)
eksctl utils associate-iam-oidc-provider \
  --region ap-northeast-1 \
  --cluster graphapp-eks-cluster \
  --approve

# Create IAM policy and Service Account for AWS LB controller (Replace ACCOUNT_ID)
curl -o iam_policy_main_branch.json https://raw.githubusercontent.com/kubernetes-sigs/aws-load-balancer-controller/main/docs/install/iam_policy.json # main
aws iam create-policy \
  --policy-name AWSLoadBalancerControllerIAMPolicy \
  --policy-document file://iam_policy_main_branch.json
eksctl create iamserviceaccount \
  --cluster graphapp-eks-cluster \
  --namespace kube-system \
  --name aws-load-balancer-controller \
  --role-name AmazonEKSLoadBalancerControllerRole \
  --attach-policy-arn arn:aws:iam::<ACCOUNT_ID>:policy/AWSLoadBalancerControllerIAMPolicy \
  --approve

# Create IAM policy and Service Account for Node AutoScaler (Replace ACCOUNT_ID)
aws iam create-policy \
  --policy-name AmazonEKSClusterAutoscalerPolicy \
  --policy-document file://cluster-autoscaler-policy.json # Refer to Appendix (a)
eksctl create iamserviceaccount \
  --cluster graphapp-eks-cluster \
  --namespace kube-system \
  --name amazon-eks-cluster-autoscaler \
  --role-name AmazonEKSClusterAutoscalerRole \
  --attach-policy-arn arn:aws:iam::aws:policy/AmazonEKSClusterAutoscalerPolicy \
  --approve

# Tag ASG to enable auto-discovery mode
aws autoscaling describe-auto-scaling-groups \
  --query 'AutoScalingGroups[].AutoScalingGroupName' # display YOUR_REAL_ASG_NAME
aws autoscaling create-or-update-tags --tags \
  ResourceId=<YOUR_REAL_ASG_NAME>,ResourceType=auto-scaling-group,Key=k8s.io/cluster-autoscaler/enabled,Value=true,PropagateAtLaunch=true \
  ResourceId=<YOUR_REAL_ASG_NAME>,ResourceType=auto-scaling-group,Key=k8s.io/cluster-autoscaler/graphapp-eks-cluster,Value=owned,PropagateAtLaunch=true

# Install autoscaler
kubectl apply -f k8s/initials/autoscaler/cluster-autoscaler-rbac.yaml
kubectl apply -f k8s/initials/autoscaler/deployment-eks-autoscaler.yaml

# Check the AutoScaler
kubectl logs -n kube-system -l app=cluster-autoscaler

# Install controller　(Replace REGION)
helm install aws-load-balancer-controller eks/aws-load-balancer-controller \
  --version 1.13.4 \
  --set clusterName=graphapp-eks-cluster \
  --set serviceAccount.create=false \
  --set serviceAccount.name=aws-load-balancer-controller \
  --set region=<REGION> \
  -n kube-system

# Check the controller
kubectl get pods -n kube-system
kubectl logs -n kube-system \
  -l app.kubernetes.io/name=aws-load-balancer-controller

# Apply ingress.yaml (ALB should immediately be created)
kubectl apply -f ./k8s/ingress/

# Check the ingresskubectl get ingress -n graphapp
kubectl get ingress -n graphapp　# ADDRESS of the created ALB should appear
kubectl describe ingress graphapp-ingress -n graphapp

# Check the healthcheck URL for frontend & backend pod (/health)
# Restart frontend pods (Replace FRONTEND POD)
kubectl get pods -n graphapp # display the name of frontend pods
kubectl delete pod <FRONTEND POD> -n graphapp

# Check in which node pods are running (Replace NODE_NAME)
kubectl get nodes
kubectl get pods --all-namespaces --field-selector spec.nodeName=<NODE_NAME>

# Update the API_BASE in the configmap-frontend.yaml (ALB endpoint)

# Apply configmap_frontend.yaml
kubectl apply -f ./k8s/initials/configmap_frontend.yaml
```

 - Commands for debugging
```
# Commands to unstall ingress (Use when re-installing)
kubectl delete -f k8s/ingress/ingress.yaml
kubectl get ingress -n graphapp # check

# Commands to unstall AWS LB controller (Use when re-installing)
helm uninstall aws-load-balancer-controller -n kube-system
kubectl get pods -n kube-system # check

# Commands to unstall IRSA (Use when re-installing)
eksctl delete iamserviceaccount \
  --cluster graphapp-eks-cluster \
  --namespace kube-system \
  --name aws-load-balancer-controller \
  --wait

kubectl get sa aws-load-balancer-controller -n kube-system # check IRSA
aws iam list-roles | grep AmazonEKSLoadBalancerControllerRole # check IAM role
aws iam list-policies --scope Local | grep AWSLoadBalancerControllerIAMPolicy # check IAM policy
```

## STEP 8 Database Setup
##########
 - If you have a snapshot, restore from it (graphapp-rds-mysql-a)
   and reset the Master password via AWS manage console using a password of your choice
 - If you need to migrate youor local DB, first create RDS instance (t3.micro/t3.small graphapp-rds-mysql-a in my case)
   and make it accessible from your local device (by making the RDS instancde public, etc.)
 - Allow inbound traffic from your working laptop's GIP* to port 3306 with Security Group
   * curl https://checkip.amazonaws.com # Check your GIP
     nc -zv <rds-endpoint> 3306 # Check connectivity
     mysql -h <rds-endpoint> -u <user> -p # log in to your RDS instance
 - Export your local database:
```
mysqldump -u graphapiuser -p GraphAPI > backup_graphapi.sql
```
 - Import into RDS:
```
mysql -h <rds-endpoint> -u <user> -p -e "CREATE DATABASE GraphAPI;"
mysql -h <rds-endpoint> -u <user> -p GraphAPI < dbbackup.sql
```

##########
## STEP 9 Ensure all required Security Groups are created and attached
##########
 -To ensure these SG are created, add the following annotation to ingress.yaml
  ALB: Client GIP -> 80/http (attached when via ingress.yaml)
  EKS nodes 1: automatically attached by EKS
  EKS nodes 2: the SG that wuold be source in the SG for RDS (no need to permit any inbound traffic) (attached via cluster-config.yaml)
  RDS: EKS nodes 2 SG -> 3306/tcp (attached when creating RDS instance)
```

# Delete the entire EKS cluster

  --name <your-nodegroup-name> \
  --region <region>
```
